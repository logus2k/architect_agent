<script>
    /**
     * UI Content Translations
    */
    const translations = {
        en: {
            menuTitle: "Our Menu",
            locationTitle: "Our Location",
            contactTitle: "Contact Us",
            home: "Home",
            sendBtn: "Send Message",
            loading: "Loading content...",
            success: "Message sent successfully! Thank you."
        },
        pt: {
            menuTitle: "Nosso Menu",
            locationTitle: "Nossa Localização",
            contactTitle: "Contacte-nos",
            home: "Início",
            sendBtn: "Enviar Mensagem",
            loading: "Carregando conteúdo...",
            success: "Mensagem enviada com sucesso! Obrigado."
        }
    };

    const langSelector = document.getElementById('langSelector');
    const menuList = document.getElementById('menuList');
    const addressDisplay = document.getElementById('addressDisplay');
    const mapFrame = document.getElementById('mapFrame');
    const contactForm = document.getElementById('contactForm');
    const formFeedback = document.getElementById('formFeedback');
    const homeLink = document.getElementById('homeLink');

    /**
     * Updates static text based on selected language
     */
    function updateUILanguage() {
        const lang = langSelector.value;
        document.getElementById('menuTitle').textContent = translations[lang].menuTitle;
        document.getElementById('locationTitle').textContent = translations[lang].locationTitle;
        document.getElementById('contactTitle').textContent = translations[lang].contactTitle;
        homeLink.textContent = translations[lang].home;
        document.getElementById('submitBtn').textContent = translations[lang].sendBtn;
        
        // Re-fetch menu when language changes
        fetchMenu(lang);
    }

    /**
     * Fetches the localized menu from the backend
     */
    async function fetchMenu(lang) {
        menuList.innerHTML = `<p class="loading">${translations[lang].loading}</p>`;
        try {
            const response = await fetch(`/public_display/get_localized_menu?language=${lang}`, {
                method: 'POST'
            });
            const data = await response.json();
            
            if (!data || !Array.isArray(data)) {
                menuList.innerHTML = '<p>No items found.</p>';
                return;
            }

            menuList.innerHTML = '';
            data.forEach(item => {
                const div = document.createElement('div');
                div.className = 'menu-item';
                
                const name = item.name || 'Unnamed Item';
                const price = item.price ? `<span class="price">${item.price}</span>` : '';
                const desc = item.description ? `<p class="description">${item.description}</p>` : '';
                
                div.innerHTML = `
                    <h3>${name} ${price}</h3>
                    ${desc}
                `;
                menuList.appendChild(div);
            });
        } catch (err) {
            menuList.innerHTML = '<p>Error loading menu.</p>';
            console.error(err);
        }
    }

    /**
     * Fetches the public location and updates the map
     */
    async function fetchLocation() {
        try {
            const response = await fetch('/public_display/get_public_location', {
                method: 'POST'
            });
            const location = await response.json();
            
            if (location && location.address) {
                addressDisplay.textContent = location.address;
            }

            if (location && location.latitude && location.longitude) {
                // Constructing OSM iframe source
                // Format: https://www.openstreetmap.org/export/embed.html?bbox=min_lon,min_lat,max_lon,max_lat
                const lat = location.latitude;
                const lon = location.longitude;
                const zoom = 15;
                const bbox = `${lon - 0.01},${lat - 0.01},${lon + 0.01},${lat + 0.01}`;
                
                mapFrame.src = `https://www.openstreetmap.org/export/embed.html?bbox=${bbox}&layer=mapnik`;
                mapFrame.style.display = 'block';
            }
        } catch (err) {
            addressDisplay.textContent = "Location information currently unavailable.";
            console.error(err);
        }
    }

    /**
     * Handles the contact form submission
     */
    contactForm.addEventListener('submit', async (e) => {
        e.preventDefault();
        
        const formData = new FormData(contactForm);
        const params = new URLSearchParams();
        params.append('userName', formData.get('userName'));
        params.append('userEmail', formData.get('userEmail'));
        params.append('message', formData.get('message'));

        try {
            const response = await fetch(`/public_display/submit_contact_form?${params.toString()}`, {
                method: 'POST'
            });

            if (response.ok) {
                formFeedback.textContent = translations[langSelector.value].success;
                formFeedback.className = 'success';
                formFeedback.style.display = 'block';
                
                // Discard/Clear the form
                contactForm.reset();
                
                // Hide success message after 5 seconds
                setTimeout(() => {
                    formFeedback.style.display = 'none';
                }, 5000);
            } else {
                throw new Error('Submission failed');
            }
        } catch (err) {
            formFeedback.textContent = "An error occurred. Please try again.";
            formFeedback.className = 'error';
            formFeedback.style.display = 'block';
            console.error(err);
        }
    });

    // Initialization
    window.addEventListener('DOMContentLoaded', () => {
        const initialLang = langSelector.value;
        updateUILanguage();
        fetchLocation();
        
        // Language change listener
        langSelector.addEventListener('change', updateUILanguage);
    });
</script>
