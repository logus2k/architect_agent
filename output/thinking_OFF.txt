    <script>
        let currentLang = 'EN';

        const translations = {
            EN: {
                menuTitle: "Our Menu",
                contactTitle: "Contact Us",
                lblName: "Name",
                lblEmail: "Email",
                lblMessage: "Message",
                btnSubmit: "Send Message"
            },
            PT: {
                menuTitle: "Nosso Menu",
                contactTitle: "Contacte-nos",
                lblName: "Nome",
                lblEmail: "E-mail",
                lblMessage: "Mensagem",
                btnSubmit: "Enviar Mensagem"
            }
        };

        async function fetchMenu() {
            const contentDiv = document.getElementById('menu-content');
            try {
                const response = await fetch(`/public_display/get_localized_menu?language=${currentLang}`, {
                    method: 'POST'
                });
                const data = await response.json();
                
                if (!data || data.length === 0) {
                    contentDiv.innerHTML = "No items available.";
                    return;
                }

                contentDiv.innerHTML = '';
                data.forEach(item => {
                    const itemDiv = document.createElement('div');
                    itemDiv.className = 'menu-item';
                    
                    const name = item.name || 'Unnamed Item';
                    const description = item.description || '';
                    const price = item.price ? ` - ${item.price}` : '';

                    itemDiv.innerHTML = `
                        <h3>${name}${price}</h3>
                        <p>${description}</p>
                    `;
                    contentDiv.appendChild(itemDiv);
                });
            } catch (error) {
                contentDiv.innerHTML = "Error loading menu.";
                console.error(error);
            }
        }

        async function fetchLocation() {
            const mapWrapper = document.getElementById('map-wrapper');
            try {
                const response = await fetch(`/public_display/get_public_location`, {
                    method: 'POST'
                });
                const data = await response.json();
                
                if (data && data.latitude && data.longitude) {
                    const lat = data.latitude;
                    const lng = data.longitude;
                    mapWrapper.innerHTML = `<iframe src="https://www.openstreetmap.org/export/embed.html?bbox=${lng-0.01}%2C${lat-0.01}%2C${lng+0.01}%2C${lat+0.01}" allowfullscreen="" loading="lazy"></iframe>`;
                } else {
                    mapWrapper.innerHTML = "Location data unavailable.";
                }
            } catch (error) {
                mapWrapper.innerHTML = "Error loading map.";
                console.error(error);
            }
        }

        function switchLanguage(lang) {
            currentLang = lang;
            
            // Update UI buttons
            document.getElementById('btn-en').classList.toggle('active', lang === 'EN');
            document.getElementById('btn-pt').classList.toggle('active', lang === 'PT');

            // Update Text
            const t = translations[lang];
            document.getElementById('menu-title').innerText = t.menuTitle;
            document.getElementById('contact-title').innerText = t.contactTitle;
            document.getElementById('label-name').innerText = t.lblName;
            document.getElementById('label-email').innerText = t.lblEmail;
            document.getElementById('label-message').innerText = t.lblMessage;
            document.getElementById('btn-submit').innerText = t.btnSubmit;

            fetchMenu();
        }

        document.getElementById('contactForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const btn = document.getElementById('btn-submit');
            const userName = document.getElementById('userName').value;
            const userEmail = document.getElementById('userEmail').value;
            const message = document.getElementById('message').value;

            btn.disabled = true;
            btn.innerText = "Sending...";

            try {
                const response = await fetch(`/public_display/submit_contact_form?userName=${encodeURIComponent(userName)}&userEmail=${encodeURIComponent(userEmail)}&message=${encodeURIComponent(message)}`, {
                    method: 'POST'
                });
                
                // Requirement: Discard the contact form submission after processing
                document.getElementById('contactForm').reset();
                alert("Message sent successfully!");
            } catch (error) {
                alert("Error sending message.");
                console.error(error);
            } finally {
                btn.disabled = false;
                btn.innerText = translations[currentLang].btnSubmit;
            }
        });

        // Initial Load
        window.onload = () => {
            fetchMenu();
            fetchLocation();
            document.getElementById('btn-en').classList.add('active');
        };
