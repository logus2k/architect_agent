    ```python
    from typing import List, Optional, Dict, Any
    from fastapi import HTTPException
    from app.repositories.store import repo

    def create_translation(itemId: str, language: str, text: str) -> Dict[str, Any]:
        if not itemId or not language or not text:
            raise ValueError("itemId, language, and text are required.")
        try:
            repository = repo("translations")
            return repository.create(itemId=itemId, language=language, text=text)
        except Exception as e:
            raise HTTPException(status_code=500, detail=f"Error creating translation: {str(e)}")

    def get_translation(id: str) -> Dict[str, Any]:
        try:
            repository = repo("translations")
            translation = repository.get(id)
            if not translation:
                raise HTTPException(status_code=404, detail="Translation not found")
            return translation
        except Exception as e:
            if isinstance(e, HTTPException): raise e
            raise HTTPException(status_code=500, detail=str(e))

    def update_translation(translation: Dict[str, Any]) -> Dict[str, Any]:
        t_id = translation.get("id")
        if not t_id:
            raise ValueError("Translation ID is required for update.")
        try:
            repository = repo("translations")
            return repository.update(t_id, translation)
        except Exception as e:
            raise HTTPException(status_code=500, detail=f"Error updating translation: {str(e)}")

    def list_translations(itemId: str) -> List[Dict[str, Any]]:
        try:
            repository = repo("translations")
            return repository.list_by_item(itemId)
        except Exception as e:
            raise HTTPException(status_code=500, detail=f"Error listing translations: {str(e)}")

    def delete_translation(id: str) -> bool:
        try:
            repository = repo("translations")
            return repository.delete(id)
        except Exception as e:
            raise HTTPException(status_code=500, detail=f"Error deleting translation: {str(e)}")
    ```
